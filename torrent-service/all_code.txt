func main() {
	cfg, err := torrentconfig.Load()
	if err != nil {
		log.Fatal(err)
	}
	l := slog.New(slog.NewTextHandler(os.Stdout, nil))

	dsn := db.BuildDSN(&cfg.Postgres)

	m, err := migrate.New("file://migrations", dsn)
	if err != nil {
		l.Error("error creating migration", slog.String("error", err.Error()))
		os.Exit(1)
	}

	if len(os.Args) < 2 {
		l.Error("add command")
		os.Exit(1)
	}
	command := os.Args[1]
	switch command {
	case "up":
		err = m.Up()
	case "down":
		err = m.Down()
	case "step":
		if len(os.Args) > 2 && os.Args[2] != "" {
			steps, parseErr := strconv.Atoi(os.Args[2])
			if parseErr != nil {
				l.Error("error converting steps to int", slog.String("error", parseErr.Error()))
				os.Exit(1)
			}
			err = m.Steps(steps)
		} else {
			l.Error("add number of steps")
			os.Exit(1)
		}
	default:
		l.Error("wrong command")
		os.Exit(1)
	}
	if err != nil {
		if errors.Is(err, migrate.ErrNoChange) {
			l.Info("database is already up")
		} else {
			l.Error("error doing migration", slog.String("error", err.Error()))
			os.Exit(1)
		}
	}

type App struct {
	cfg        *torrentconfig.Config
	l          logger.Logger
	pool       *pgxpool.Pool
	grpcServer *grpc.Server
	httpServer *http.Server

func New(ctx context.Context) (*App, error) {
	cfg, err := torrentconfig.Load()
	if err != nil {
		return nil, fmt.Errorf("init config: %w", err)
	}

	l, err := logger.New(&cfg.Logger)
	if err != nil {
		return nil, fmt.Errorf("init logger: %w", err)
	}
	l.Info(ctx, "config & logger init complete")

	// создаёт пул и делает Ping
	pool, err := db.NewPool(&cfg.Postgres)
	if err != nil {
		return nil, fmt.Errorf("create db pool: %w", err)
	}
	l.Info(ctx, "db pool init complete")

	repo := repository.NewPostgresRepository(pool)
	catrepo := repository.NewCategoriesRepository(pool)
	svc := service.NewTorrentService(repo, catrepo)
	handler := transport.NewServer(svc)

	grpcServer := grpc.NewServer(
		grpc.ChainUnaryInterceptor(
			interceptors.LoggingInterceptor(l),
			interceptors.RecoveryInterceptor(l),
			interceptors.TimeoutInterceptor(&cfg.Grpc),
			interceptors.BlockForBannedInterceptor(),
		))
	pb.RegisterTorrentServiceServer(grpcServer, handler)

	httpServer := infraServerInit(pool, &cfg.Infra, l)

	return &App{
		cfg:        cfg,
		l:          l,
		pool:       pool,
		grpcServer: grpcServer,
		httpServer: httpServer,
	}, nil
}

func (a *App) Run(ctx context.Context) error {
	logCtx := context.WithoutCancel(ctx)

	g, gctx := errgroup.WithContext(ctx)

	// gRPC-сервер
	g.Go(func() error {
		lis, err := net.Listen("tcp", fmt.Sprintf(":%d", a.cfg.Grpc.Port))
		if err != nil {
			return fmt.Errorf("grpc listen: %w", err)
		}
		a.l.Info(logCtx, "grpc server starting", logger.Int("port", a.cfg.Grpc.Port))

		err = a.grpcServer.Serve(lis)
		if errors.Is(err, grpc.ErrServerStopped) {
			return nil
		}
		return err
	})

	// http-сервер (health, ready, metrics)
	g.Go(func() error {
		a.l.Info(logCtx, "http infra server starting", logger.Int("port", a.cfg.Infra.Port))

		err := a.httpServer.ListenAndServe()
		if errors.Is(err, http.ErrServerClosed) {
			return nil // после Shutdown это не ошибка
		}
		return err
	})

	// Сторож: спит, пока не придёт сигнал или не упадёт один из серверов
	g.Go(func() error {
		<-gctx.Done()
		return a.shutdown(logCtx)
	})

	if err := g.Wait(); err != nil {
		a.l.Error(logCtx, "torrent-service stopped with error", logger.Error(err))
		return err
	}
	a.l.Info(logCtx, "torrent-service stopped gracefully")
	return nil
}

// shutdown останавливает оба сервера одновременно
func (a *App) shutdown(ctx context.Context) error {
	a.l.Info(ctx, "shutting down")

	timeout, cancel := context.WithTimeout(context.Background(), a.cfg.GracefulShutdownTimeout)
	defer cancel()

	grpcStopped := make(chan struct{})
	httpStopped := make(chan struct{})
	allStopped := make(chan struct{})
	

	go func() {
		defer close(grpcStopped)
		a.grpcServer.GracefulStop()

	}()
	var httpErr error
	go func() {
		defer close(httpStopped)
		httpErr = a.httpServer.Shutdown(timeout)
	}()
	go func() {
		<-grpcStopped
		<-httpStopped
		close(allStopped)
	}()

	select {
	case <-allStopped:
		if httpErr != nil {
			a.httpServer.Close()
			return fmt.Errorf("http shutdown: %w", httpErr)
		}
		a.l.Info(ctx, "grpc and http servers stopped gracefully")
		return nil
	case <-timeout.Done():
		a.l.Error(ctx, "shutdown timed out, forcing stop")
		a.grpcServer.Stop()
		a.httpServer.Close()
		return errors.New("shutdown timeout: servers were stopped forcibly")
	}
}

func (a *App) Close() error {
	a.pool.Close()

	a.l.Sync()

	return nil

package app

import (
	"context"
	"fmt"
	"net/http"

	"github.com/jackc/pgx/v5/pgxpool"
	"github.com/monstrong/gracker2/pkg/config"
	"github.com/monstrong/gracker2/pkg/logger"
	"github.com/prometheus/client_golang/prometheus/promhttp"
)

func infraServerInit(pool *pgxpool.Pool, cfg *config.Infra, l logger.Logger) *http.Server {
	mux := http.NewServeMux()

	mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
		w.Write([]byte("OK"))
	})

	mux.HandleFunc("/ready", func(w http.ResponseWriter, r *http.Request) {

		ctx, cancel := context.WithTimeout(r.Context(), cfg.Timeout/2)
		defer cancel()

		if err := pool.Ping(ctx); err != nil {
			w.WriteHeader(http.StatusServiceUnavailable)
			w.Write([]byte("ERROR"))
			l.Error(ctx, "error on connection to db", logger.Error(err))
			
		} else {
			w.WriteHeader(http.StatusOK)
			w.Write([]byte("OK"))
		}
	})

	mux.Handle("/metrics", promhttp.Handler())

	return &http.Server{
		Addr:    fmt.Sprintf(":%d", cfg.Port),
		Handler: mux,
		ReadHeaderTimeout: cfg.Timeout,
		WriteTimeout:      cfg.Timeout,
		ReadTimeout:       cfg.Timeout,
		IdleTimeout:       cfg.Timeout*6,
	}
}

func (s *TorrentService) validateTorrent(ctx context.Context, torrent *models.Torrent) error {
	if err := validateName(torrent.Name); err != nil {
		return err
	}
	if err := validateDescription(torrent.Description); err != nil {

	if err := validateInfoHash(torrent.InfoHash); err != nil {

	}

		return err
	}

		return err

		return err
	}
	return nil
}

// other roles are blocked by interceptor
func validateRole(role domain.Role) error {
	switch role {
	case domain.UserRole,
		domain.ModeratorRole,
		domain.AdminRole,
		domain.OwnerRole:
		return nil
	default:
		return &errs.ValidationError{Field: "role", Reason: "unknown role"}
	}
}

func validateID(id uuid.UUID) error {
	if id == uuid.Nil {
		return &errs.ValidationError{Field: "id", Reason: "invalid torrent id format"}
	}
	return nil
}

func validateName(name string) error {
	if name == "" {
		return &errs.ValidationError{Field: "name", Reason: "'name' is required"}
	}
	if len(name) > nameMaxLen {
		return &errs.ValidationError{Field: "name", Reason: "'name' is too long"}
	}
	return nil
}

func validateDescription(description string) error {
	if description == "" {
		return &errs.ValidationError{Field: "description", Reason: "'description' is required"}
	}
	if len(description) > descriptionMaxLen {
		return &errs.ValidationError{Field: "description", Reason: "'description' is too long"}
	}
	return nil
}

func validateInfoHash(infoHash string) error {
	if infoHash == "" {
		return &errs.ValidationError{Field: "infoHash", Reason: "'infoHash' is required"}
	}
	return nil

func (s *TorrentService) validateTorrentPatch(ctx context.Context, patch *models.TorrentPatch) error {
	if patch == nil {
		return &errs.ValidationError{Field: "patch", Reason: "'patch' is required"}
	}
	if patch.CategoryID != nil && patch.Description == nil && patch.Name == nil && patch.InfoHash == nil {
		return &errs.ValidationError{Field: "patch", Reason: "at least one field is required"}
	}
	if patch.CategoryID != nil {
		if err := s.checkCategoryID(ctx, *patch.CategoryID); err != nil {
			return err
		}
	}
	if patch.Name != nil {
		if err := validateName(*patch.Name); err != nil {
			return err
		}
	}
	if patch.Description != nil {
		if err := validateDescription(*patch.Description); err != nil {
			return err
		}
	}
	if patch.InfoHash != nil {
		if err := validateInfoHash(*patch.InfoHash); err != nil {
			return err
		}
	}
	return nil
}

func (r *PostgresRepository) Update(ctx context.Context, patch *models.TorrentPatch, id uuid.UUID) (*models.Torrent, error) {
	const op = "repo.postgres.update"

	b := psql.Update("torrents")

	}
	if patch.Description != nil {
		b = b.Set("description", *patch.Description)

	if patch.InfoHash != nil {
		b = b.Set("info_hash", *patch.InfoHash)
	}

	b = b.Set("updated_at", sq.Expr("CURRENT_TIMESTAMP"))
	b = b.Where(sq.Eq{"id": id})
	b = b.Suffix(fmt.Sprintf(`RETURNING %s`, torrentColumns))

	query, args, err := b.ToSql()
	if err != nil {
		return nil, fmt.Errorf("%s: %w", op, err)
	}

	var t models.Torrent
	err = r.db.QueryRow(ctx, query, args...).Scan(
		&t.ID,
		&t.Name,
		&t.Description,
		&t.InfoHash,
		&t.Status,
		&t.AuthorID,
		&t.CategoryID,
		&t.Downloads,
		&t.CreatedAt,
		&t.UpdatedAt,
	)
	if err != nil {
		if errors.Is(err, pgx.ErrNoRows) {
			return nil, fmt.Errorf("%s: %w", op, errs.ErrNotFound)

func checkIDfromMD(MD metadata.MD) (userID uuid.UUID, err error) {

			return uuid.Nil, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyUserID, Reason: "invalid user id format"}
		}
	} else {
		return uuid.Nil, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyUserID, Reason: "empty user id"}
	}

	return
}

func checkSenderRolefromMD(MD metadata.MD) (SenderRole domain.Role, err error) {
	vals := MD.Get(grpcmetakey.GRPCMetaKeyRole)

	if len(vals) > 0 {

		if err != nil {
			return domain.RoleUnspecified, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyRole, Reason: "invalid role format"}
		}
		SenderRole = domain.Role(roleValue)
	} else {
		return domain.RoleUnspecified, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyRole, Reason: "empty role"}
	}
