// App собирает сервис из частей, запускает его и освобождает ресурсы.
type App struct {
	cfg        *config.Config
	l          logger.Logger
	pool       *pgxpool.Pool
	grpcServer *grpc.Server
	httpServer *http.Server
}

// New создаёт все части сервиса, но ничего не запускает.
func New(ctx context.Context) (*App, error) {
	cfg, err := config.Load()
	if err != nil {
		return nil, fmt.Errorf("init config: %w", err)
	}

	l, err := logger.New(&cfg.Logger)
	if err != nil {
		return nil, fmt.Errorf("init logger: %w", err)
	}
	l.Info(ctx, "config & logger init complete")

	// создаёт пул и делает Ping
	pool, err := db.NewPool(&cfg.Postgres)
	if err != nil {
		return nil, fmt.Errorf("create db pool: %w", err)
	}
	l.Info(ctx, "db pool init complete")

	repo := repository.NewPostgresRepository(pool)
	svc := service.NewTorrentService(repo)
	handler := transport.NewServer(svc)

	grpcServer := grpc.NewServer(
		grpc.ChainUnaryInterceptor(
			interceptors.RecoveryInterceptor(l),
			interceptors.LoggingInterceptor(l),
			interceptors.TimeoutInterceptor(&cfg.Grpc),
		))
	pb.RegisterTorrentServiceServer(grpcServer, handler)

	httpServer := infraServerInit(pool, cfg, l)

	return &App{
		cfg:        cfg,
		l:          l,
		pool:       pool,
		grpcServer: grpcServer,
		httpServer: httpServer,
	}, nil
}

func (a *App) Run(ctx context.Context) error {
	logCtx := context.WithoutCancel(ctx)

	g, gctx := errgroup.WithContext(ctx)

	// gRPC-сервер
	g.Go(func() error {
		lis, err := net.Listen("tcp", fmt.Sprintf(":%d", a.cfg.Grpc.Port))
		if err != nil {
			return fmt.Errorf("grpc listen: %w", err)
		}
		a.l.Info(logCtx, "grpc server starting", logger.Int("port", a.cfg.Grpc.Port))

		err = a.grpcServer.Serve(lis)
		if errors.Is(err, grpc.ErrServerStopped) {
			return nil // остановка пришла раньше старта: это не ошибка
		}
		return err
	})

	// http-сервер (health, ready, metrics)
	g.Go(func() error {
		a.l.Info(logCtx, "http infra server starting")

		err := a.httpServer.ListenAndServe()
		if errors.Is(err, http.ErrServerClosed) {
			return nil // после Shutdown это не ошибка
		}
		return err
	})

	// Сторож: спит, пока не придёт сигнал или не упадёт один из серверов
	g.Go(func() error {
		<-gctx.Done()
		return a.shutdown(logCtx)
	})

	if err := g.Wait(); err != nil {
		a.l.Error(logCtx, "torrent-service stopped with error", logger.Error(err))
		return err
	}
	a.l.Info(logCtx, "torrent-service stopped gracefully")
	return nil
}

// shutdown останавливает оба сервера одновременно, на всё даётся 15 секунд.
func (a *App) shutdown(ctx context.Context) error {
	a.l.Info(ctx, "shutting down")

	// Родитель Background: контекст от сигнала к этому моменту уже отменён.
	timeout, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()

	grpcStopped := make(chan struct{})
	httpStopped := make(chan struct{})
	allStopped := make(chan struct{})

	// httpErr записывается до close(httpStopped), а читается только после allStopped.
	var httpErr error

	go func() {
		defer close(grpcStopped)
		a.grpcServer.GracefulStop()

	}()
	go func() {
		defer close(httpStopped)
		httpErr = a.httpServer.Shutdown(timeout)
	}()
	go func() {
		<-grpcStopped
		<-httpStopped
		close(allStopped)
	}()

	select {
	case <-allStopped:
		if httpErr != nil {
			a.httpServer.Close()
			return fmt.Errorf("http shutdown: %w", httpErr)
		}
		a.l.Info(ctx, "grpc and http servers stopped gracefully")
		return nil
	case <-timeout.Done():
		a.l.Error(ctx, "shutdown timed out, forcing stop")
		a.grpcServer.Stop()
		a.httpServer.Close()
		return errors.New("shutdown timeout: servers were stopped forcibly")
	}
}

// Close освобождает ресурсы. Вызывать после Run.
func (a *App) Close() error {
	a.pool.Close()

	a.l.Sync()

	return nil

)

func infraServerInit(pool *pgxpool.Pool, cfg *config.Config, l logger.Logger) *http.Server {
	mux := http.NewServeMux()

	mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
		w.Write([]byte("OK"))
	})

	mux.HandleFunc("/ready", func(w http.ResponseWriter, r *http.Request) {

		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()

		if err := pool.Ping(ctx); err != nil {
			w.WriteHeader(http.StatusServiceUnavailable)
			w.Write([]byte("ERROR"))
			l.Error(ctx, "error on connection to db", logger.Error(err))
			
		} else {
			w.WriteHeader(http.StatusOK)
			w.Write([]byte("OK"))
		}
	})

	mux.Handle("/metrics", promhttp.Handler())

	return &http.Server{
		Addr:    fmt.Sprintf(":%d", cfg.Infra.Port),
		Handler: mux,
	}
}

		}
		return nil, fmt.Errorf("%s: %w", op, err)
	}
	return t, nil

}

func (r *PostgresRepository) List(ctx context.Context, limit, offset int) ([]*models.Torrent, error) {
	const op = "repo.postgres.list"
	var ts []*models.Torrent
	query := `SELECT id, name, description, info_hash, status, author_id, category_id, downloads, created_at, updated_at FROM torrents 
	ORDER BY created_at DESC LIMIT $1 OFFSET $2`
	rows, err := r.db.Query(ctx, query, limit, offset)
	if err != nil {
		return nil, fmt.Errorf("%s: %w", op, err)
	}
	defer rows.Close()

	for rows.Next() {
		var t models.Torrent
		err := rows.Scan(
			&t.ID,
			&t.Name,
			&t.Description,
			&t.InfoHash,
			&t.Status,
			&t.AuthorID,
			&t.CategoryID,
			&t.Downloads,
			&t.CreatedAt,
			&t.UpdatedAt,
		)
		if err != nil {
			return nil, fmt.Errorf("%s: %w", op, err)
		}
		ts = append(ts, &t)
	}
	if err := rows.Err(); err != nil {
		return nil, fmt.Errorf("%s: %w", op, err)
	}
	return ts, nil
}

	return &pb.ListTorrentsResponse{Torrents: pbTorrents}, nil
}

func (s *Server) CreateTorrent(ctx context.Context, in *pb.CreateTorrentRequest) (*pb.CreateTorrentResponse, error) {
	authorID, err := uuid.Parse(in.GetAuthorId())
	if err != nil {
		return nil, &errs.ValidationError{Field: "authorID", Reason: "invalid authorID id format"}
	}
	categoryID, err := uuid.Parse(in.GetCategoryId())
	if err != nil {
		return nil, &errs.ValidationError{Field: "categoryID", Reason: "invalid categoryID id format"}
	}
	t := models.Torrent{Name: in.GetName(), 
		Description: in.GetDescription(), 
		InfoHash: in.GetInfoHash(), 
		AuthorID: authorID,
		CategoryID: categoryID,
	}
	m, err := s.s.Create(ctx, &t)
	if err != nil{
		return nil, err
	}

	return &pb.CreateTorrentResponse{Torrent: toProto(m)}, nil
}


func (s *Server) DeleteTorrent(ctx context.Context, in *pb.DeleteTorrentRequest) (*pb.DeleteTorrentResponse, error) {
	id, err := uuid.Parse(in.GetId())
	if err != nil {
		return nil, &errs.ValidationError{Field: "id", Reason: "invalid torrent id format"}
	}

	ok, err := s.s.Delete(ctx, id)
	if err != nil {
		return nil, err
	}
	return &pb.DeleteTorrentResponse{Deleted: ok}, nil
}

		Downloads: int64(m.Downloads),
		CreatedAt: timestamppb.New(m.CreatedAt),
		UpdatedAt: timestamppb.New(m.UpdatedAt),
	}
