func main() {
	cfg, err := config.Load()
	if err != nil {
		log.Fatal(err)
	}
	l := slog.New(slog.NewTextHandler(os.Stdout, nil))

	dsn := db.BuildDSN(&cfg.Postgres)

	m, err := migrate.New("file://migrations", dsn)
	if err != nil {
		l.Error("error creating migration", slog.String("error", err.Error()))
		os.Exit(1)
	}

	if len(os.Args) < 2 {
		l.Error("add command")
		os.Exit(1)
	}
	command := os.Args[1]
	switch command {
	case "up":
		err = m.Up()
	case "down":
		err = m.Down()
	case "step":
		if len(os.Args) > 2 && os.Args[2] != "" {
			steps, parseErr := strconv.Atoi(os.Args[2])
			if parseErr != nil {
				l.Error("error converting steps to int", slog.String("error", parseErr.Error()))
				os.Exit(1)
			}
			err = m.Steps(steps)
		} else {
			l.Error("add number of steps")
			os.Exit(1)
		}
	default:
		l.Error("wrong command")
		os.Exit(1)
	}
	if err != nil {
		if errors.Is(err, migrate.ErrNoChange) {
			l.Info("database is already up")
		} else {
			l.Error("error doing migration", slog.String("error", err.Error()))
			os.Exit(1)
		}
	}
}

func HashPassword(password string, cost int) (string, error) { 
	hash, err := bcrypt.GenerateFromPassword([]byte(password), cost)
	if err != nil {
		return "", err
	}
	return string(hash), nil

type App struct {
	cfg        *authconfig.Config
	l          logger.Logger
	pool       *pgxpool.Pool
	grpcServer *grpc.Server
	httpServer *http.Server
}

func New(ctx context.Context) (*App, error) {
	cfg, err := authconfig.Load()
	if err != nil {
		return nil, fmt.Errorf("init config: %w", err)
	}

	l, err := logger.New(&cfg.Logger)
	if err != nil {
		return nil, fmt.Errorf("init logger: %w", err)
	}
	l.Info(ctx, "config & logger init complete")

	// создаёт пул и делает Ping
	pool, err := db.NewPool(&cfg.Postgres)
	if err != nil {
		return nil, fmt.Errorf("create db pool: %w", err)
	}
	l.Info(ctx, "db pool init complete")

	repo := repository.NewPostgresRepository(pool)
	svc := service.NewAuthService(repo)
	handler := transport.NewServer(svc)

	grpcServer := grpc.NewServer(
		grpc.ChainUnaryInterceptor(
			interceptors.RecoveryInterceptor(l),
			interceptors.LoggingInterceptor(l),
			interceptors.TimeoutInterceptor(&cfg.Grpc),
		))
	pb.RegisterAuthServiceServer(grpcServer, handler)

	httpServer := infraServerInit(pool, cfg, l)

	return &App{
		cfg:        cfg,
		l:          l,
		pool:       pool,
		grpcServer: grpcServer,
		httpServer: httpServer,
	}, nil
}

func (a *App) Run(ctx context.Context) error {
	logCtx := context.WithoutCancel(ctx)

	g, gctx := errgroup.WithContext(ctx)

	// gRPC-сервер
	g.Go(func() error {
		lis, err := net.Listen("tcp", fmt.Sprintf(":%d", a.cfg.Grpc.Port))
		if err != nil {
			return fmt.Errorf("grpc listen: %w", err)
		}
		a.l.Info(logCtx, "grpc server starting", logger.Int("port", a.cfg.Grpc.Port))

		err = a.grpcServer.Serve(lis)
		if errors.Is(err, grpc.ErrServerStopped) {
			return nil
		}
		return err
	})

	// http-сервер (health, ready, metrics)
	g.Go(func() error {
		a.l.Info(logCtx, "http infra server starting")

		err := a.httpServer.ListenAndServe()
		if errors.Is(err, http.ErrServerClosed) {
			return nil // после Shutdown это не ошибка
		}
		return err
	})

	// Сторож: спит, пока не придёт сигнал или не упадёт один из серверов
	g.Go(func() error {
		<-gctx.Done()
		return a.shutdown(logCtx)
	})

	if err := g.Wait(); err != nil {
		a.l.Error(logCtx, "auth-service stopped with error", logger.Error(err))
		return err
	}
	a.l.Info(logCtx, "auth-service stopped gracefully")
	return nil
}

// shutdown останавливает оба сервера одновременно
func (a *App) shutdown(ctx context.Context) error {
	a.l.Info(ctx, "shutting down")

	timeout, cancel := context.WithTimeout(context.Background(), a.cfg.GracefulShutdownTimeout)
	defer cancel()

	grpcStopped := make(chan struct{})
	httpStopped := make(chan struct{})
	allStopped := make(chan struct{})
	

	go func() {
		defer close(grpcStopped)
		a.grpcServer.GracefulStop()

	}()
	var httpErr error
	go func() {
		defer close(httpStopped)
		httpErr = a.httpServer.Shutdown(timeout)
	}()
	go func() {
		<-grpcStopped
		<-httpStopped
		close(allStopped)
	}()

	select {
	case <-allStopped:
		if httpErr != nil {
			a.httpServer.Close()
			return fmt.Errorf("http shutdown: %w", httpErr)
		}
		a.l.Info(ctx, "grpc and http servers stopped gracefully")
		return nil
	case <-timeout.Done():
		a.l.Error(ctx, "shutdown timed out, forcing stop")
		a.grpcServer.Stop()
		a.httpServer.Close()
		return errors.New("shutdown timeout: servers were stopped forcibly")
	}
}

func (a *App) Close() error {
	a.pool.Close()

	a.l.Sync()

	return nil

package app

import (
	"context"
	"fmt"
	"net/http"
	"time"

	"github.com/jackc/pgx/v5/pgxpool"
	"github.com/monstrong/gracker2/pkg/config"
	"github.com/monstrong/gracker2/pkg/logger"
	"github.com/prometheus/client_golang/prometheus/promhttp"
)

func infraServerInit(pool *pgxpool.Pool, cfg *config.Config, l logger.Logger) *http.Server {
	mux := http.NewServeMux()

	mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
		w.Write([]byte("OK"))
	})

	mux.HandleFunc("/ready", func(w http.ResponseWriter, r *http.Request) {

		ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
		defer cancel()

		if err := pool.Ping(ctx); err != nil {
			w.WriteHeader(http.StatusServiceUnavailable)
			w.Write([]byte("ERROR"))
			l.Error(ctx, "error on connection to db", logger.Error(err))
			
		} else {
			w.WriteHeader(http.StatusOK)
			w.Write([]byte("OK"))
		}
	})

	mux.Handle("/metrics", promhttp.Handler())

	return &http.Server{
		Addr:    fmt.Sprintf(":%d", cfg.Infra.Port),
		Handler: mux,
	}
}

import (
	"fmt"
	"os"
	"time"

	"github.com/monstrong/gracker2/pkg/config"
	"golang.org/x/crypto/bcrypt"
)

type Config struct {
	config.Config

	JWT    JWT    `mapstructure:"jwt" yaml:"jwt"`
	Hasher Hasher `mapstructure:"hasher" yaml:"hasher"`
}

type JWT struct {
	AccessTokenTTL  time.Duration `mapstructure:"access_token_ttl" yaml:"access_token_ttl"`
	RefreshTokenTTL time.Duration `mapstructure:"refresh_token_ttl" yaml:"refresh_token_ttl"`
	SigningKey      string
}

type Hasher struct {
	Cost int `mapstructure:"cost" yaml:"cost"`
}

const secretKey = "JWT_SECRET"
const postgresPasswordKey = "POSTGRES_PASSWORD"

func Load() (*Config, error) {
	cfg := &Config{}

	if err := config.LoadInto(cfg); err != nil {
		return nil, err
	}

	cfg.Postgres.Password = os.Getenv(postgresPasswordKey)
	cfg.JWT.SigningKey = os.Getenv(secretKey)

	if err := config.Validate(&cfg.Config); err != nil {
		return nil, err
	}

	if cfg.JWT.SigningKey == "" {
		return nil, fmt.Errorf("empty JWT secret")
	}
	if cfg.JWT.AccessTokenTTL <= 0 {
		return nil, fmt.Errorf("invalid access token TTL")
	}
	if cfg.JWT.RefreshTokenTTL <= 0 {
		return nil, fmt.Errorf("invalid refresh token TTL")
	}
	if cfg.Hasher.Cost < bcrypt.MinCost ||
		cfg.Hasher.Cost > bcrypt.MaxCost {
		return nil, fmt.Errorf("invalid bcrypt cost: %d", cfg.Hasher.Cost)
	}

	return cfg, nil
}
