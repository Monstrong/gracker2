type App struct {
	cfg        *gatewayconfig.Config
	httpServer *http.Server
	l          logger.Logger
	clients    []Client
}

type Client interface { // io.Closer
	Close() error
}

func New(ctx context.Context) (*App, error) {
	cfg, err := gatewayconfig.Load()
	if err != nil {
		return nil, fmt.Errorf("init config: %w", err)
	}

	l, err := logger.New(&cfg.Logger)
	if err != nil {
		return nil, fmt.Errorf("init logger: %w", err)
	}
	l.Info(ctx, "config & logger init complete")

	server := newHttpServer(&cfg.Server)

	return &App{
		cfg:        cfg,
		httpServer: server,
		l:          l,
	}, nil
}

func (a *App) Run(ctx context.Context) error {
	logCtx := context.WithoutCancel(ctx) // контекст для логгера (останется после отмены основного через gracefulShutdown)

	serveErr := make(chan error, 1)
	go func() {
		a.l.Info(logCtx, "api-gateway server starting")

		err := a.httpServer.ListenAndServe()
		if errors.Is(err, http.ErrServerClosed) {
			err = nil // штатное завершение
		}
		serveErr <- err
	}()

	select {
	case err := <-serveErr:
		if err != nil {
			a.l.Error(logCtx, "api-gateway server failed", logger.Error(err))
			return fmt.Errorf("api-gateway server: %w", err)
		}
		a.l.Info(logCtx, "api-gateway server stopped")
		return nil
	case <-ctx.Done():
		a.l.Info(logCtx, "shutdown signal received")
	}

	if err := a.shutdown(logCtx); err != nil {
		a.l.Error(logCtx, "graceful shutdown failed", logger.Error(err))
		return err
	}

	a.l.Info(logCtx, "api-gateway stopped gracefully")
	return nil
}

// graceful shutdown
func (a *App) shutdown(ctx context.Context) error {
	a.l.Info(ctx, "shutting down")

	shutCtx, cancel := context.WithTimeout(context.Background(), a.cfg.GracefulShutdownTimeout)
	defer cancel()

	if err := a.httpServer.Shutdown(shutCtx); err != nil {
		_ = a.httpServer.Close()
		return fmt.Errorf("http server shutdown: %w", err)
	}

	return nil
}
func (a *App) Close() error {
	a.l.Sync()

	var errs []error 
	// закрывает все соединения клиентов
    for _, c := range a.clients {
        if err := c.Close(); err != nil {
            errs = append(errs, err)
        }
    }
    return errors.Join(errs...)
}

func Load() (*Config, error) {
	v := viper.New()
	v.SetConfigFile(configYamlPath)
	cfg := &Config{}

	if err := v.ReadInConfig(); err != nil {
		return nil, fmt.Errorf("error reading config.yaml: %w", err)
	}

	if err := godotenv.Load(configEnvPath); errors.Is(err, os.ErrNotExist) {
		fmt.Println("no .env file found")
	}

	v.SetEnvKeyReplacer(strings.NewReplacer(".", "_"))
	v.AutomaticEnv()

	if err := v.Unmarshal(cfg); err != nil {
		return nil, fmt.Errorf("unmarshal config: %w", err)
	}	
	
	cfg.JWT_Secret = os.Getenv(secretKey)
	
	if err := cfg.validate(); err != nil {
		return nil, fmt.Errorf("invalid config: %w", err)
	}

	return cfg, nil
}

func (c *Config) validate() error {
	switch {
	// services
	case c.Services.Auth.Addr == "":
		return fmt.Errorf("invalid auth_addr: empty")
	case c.Services.Torrent.Addr == "":
		return fmt.Errorf("invalid torrent_addr: empty")
	case c.Services.Auth.Timeout <= 0 || c.Services.Auth.Timeout > time.Minute:
		return fmt.Errorf("invalid auth_timeout: %s, must be > 0 and <= 1m", c.Services.Auth.Timeout)
	case c.Services.Torrent.Timeout <= 0 || c.Services.Torrent.Timeout > time.Minute:
		return fmt.Errorf("invalid torrent_timeout: %s, must be > 0 and <= 1m", c.Services.Torrent.Timeout)

	// graceful_shutdown_timeout
	case c.GracefulShutdownTimeout <= 0 || c.GracefulShutdownTimeout > time.Minute:
		return fmt.Errorf("invalid graceful_shutdown_timeout: %s, must be > 0 and <= 1m", c.GracefulShutdownTimeout)

	// server
	case c.Server.Name == "":
		return fmt.Errorf("invalid server name: empty")
	case c.Server.Port <= 0:
		return fmt.Errorf("invalid server port: %d, must be > 0", c.Server.Port)
	case c.Server.Timeout <= 0 || c.Server.Timeout > time.Minute:
		return fmt.Errorf("invalid server timeout: %s, must be > 0 and <= 1m", c.Server.Timeout)

	// logger
	case c.Logger.Level == "":
		return fmt.Errorf("invalid logger level: empty")

	default:
		return nil
	}

func (h *AuthHandler) Login(w http.ResponseWriter, r *http.Request) {
	var req dto.LoginRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, err.Error(), http.StatusBadRequest)
		return
	}

	ctx, cancel := context.WithTimeout(r.Context(), h.cfg.Timeout)
	defer cancel()

	grpcReq := mapper.ToLoginGRPC(req)

    resp, err := h.authClient.Login(ctx, grpcReq)
    if err != nil {
        writeGRPCError(w, err)
        return
    }

    httpResp := mapper.ToLoginHTTP(resp)

    if err := writeJSON(w, http.StatusOK, httpResp); err != nil {
		// TODO: log
	}
}

func (h *AuthHandler) GetMe(w http.ResponseWriter, r *http.Request) {
	claims, ok := middleware.GetClaims(r.Context())
	if !ok {
		http.Error(w, "unauthorized", http.StatusUnauthorized)
		return
	}

	ctx, cancel := context.WithTimeout(r.Context(), h.cfg.Timeout)
	defer cancel()

	ctx = metadata.AppendToOutgoingContext(
        ctx,
        "x-user-id", claims.UserID.String(),
        "x-sender-role", strconv.Itoa(int(claims.Role)),
    )

	resp, err := h.authClient.GetMe(ctx, &authv1.GetMeRequest{})
	if err != nil {
		writeGRPCError(w, err)
		return
	}

	httpResp := mapper.ToGetMeHTTP(resp)

	if err := writeJSON(w, http.StatusOK, httpResp); err != nil {
		// TODO: log
	}
}

func writeJSON(w http.ResponseWriter, status int, data any) error {
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(status)
	return json.NewEncoder(w).Encode(data)
}
func writeGRPCError(w http.ResponseWriter, err error) {
	st, ok := status.FromError(err)
	if !ok {
		http.Error(w, "internal server error", http.StatusInternalServerError)
		return
	}
	
	switch st.Code() {
    case codes.InvalidArgument:
        http.Error(w, st.Message(), http.StatusBadRequest)
    case codes.Unauthenticated:
        http.Error(w, st.Message(), http.StatusUnauthorized)
    case codes.PermissionDenied:
        http.Error(w, st.Message(), http.StatusForbidden)
    case codes.NotFound:
        http.Error(w, st.Message(), http.StatusNotFound)
    case codes.AlreadyExists:
        http.Error(w, st.Message(), http.StatusConflict)
    case codes.DeadlineExceeded:
        http.Error(w, st.Message(), http.StatusGatewayTimeout)
    case codes.Unavailable:
        http.Error(w, st.Message(), http.StatusServiceUnavailable)
    default:
        http.Error(w, "internal server error", http.StatusInternalServerError)
    }

package middleware

import (
	"context"
	"net/http"
	"strings"

	"github.com/monstrong/gracker2/pkg/jwtparser"
)

const userClaimsKey string = "user_claims"

func AuthMiddleware(tokenManager jwtparser.TokenManager) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			authHeader := r.Header.Get("Authorization")
			parts := strings.Fields(authHeader)

			if len(parts) != 2 || !strings.EqualFold(parts[0], "Bearer") || parts[1] == "" {
				http.Error(w, "missing or invalid authorization header", http.StatusUnauthorized)
    			return
			}
			
			tokenString := parts[1]
			claims, err := tokenManager.Parse(tokenString)
			if err != nil {
				http.Error(w, "invalid token", http.StatusUnauthorized)
				return
			}
			ctx := r.Context()
			ctx = context.WithValue(ctx, userClaimsKey, claims)
			next.ServeHTTP(w, r.WithContext(ctx))
		})
	}
}

func GetClaims(ctx context.Context) (*jwtparser.Claims, bool) {
	claims, ok := ctx.Value(userClaimsKey).(*jwtparser.Claims)
	return claims, ok

	pool, err := pgxpool.NewWithConfig(ctx, poolCfg)

	}

	if err := pool.Ping(ctx); err != nil {
		pool.Close()
		return nil, fmt.Errorf("ping database: %w", err)
	}

	return pool, nil
}

const DurationKey = "request-duration"

func LoggingInterceptor(l logger.Logger) grpc.UnaryServerInterceptor {
	return func(
		ctx context.Context,
		req any,
		info *grpc.UnaryServerInfo,
		handler grpc.UnaryHandler,
	) (any, error) {
		start := time.Now()
		reqID := uuid.New().String()

		MD, _ := metadata.FromIncomingContext(ctx)
		var traceID string
		vals := MD.Get(string(logger.LoggerTraceIDKey))
		if len(vals) > 0 {
			if _, err := uuid.Parse(vals[0]); err != nil {
				l.Error(ctx, info.FullMethod, logger.Error(err))
				traceID = uuid.New().String()
			} else {
				traceID = vals[0]
			}
		} else {
			traceID = uuid.New().String()
		}

		ctx = logger.WithRequestID(ctx, reqID)
		ctx = logger.WithTraceID(ctx, traceID)
		res, err := handler(ctx, req)
		duration := time.Since(start)

		if err != nil {
			var ve *errs.ValidationError
			switch {
			case errors.As(err, &ve):
				l.Info(ctx, info.FullMethod, logger.Duration(DurationKey, duration), logger.Error(err))
			case errors.Is(err, errs.ErrNotFound),
				errors.Is(err, errs.ErrAlreadyExists),
				errors.Is(err, errs.ErrInvalidCredentials),
				errors.Is(err, errs.ErrNameAlreadyExists),
				errors.Is(err, errs.ErrEmailAlreadyExists),
				errors.Is(err, errs.ErrUserPatchEmpty),
				errors.Is(err, errs.ErrPermissionDenied),
				errors.Is(err, errs.ErrPasswordMismatch),
				errors.Is(err, errs.ErrWrongMethod),
				errors.Is(err, errs.ErrBannedUser):
				
				l.Info(ctx, info.FullMethod, logger.Duration(DurationKey, duration), logger.Error(err))
			case errors.Is(err, context.DeadlineExceeded),
				errors.Is(err, context.Canceled):
				l.Warn(ctx, info.FullMethod, logger.Duration(DurationKey, duration), logger.Error(err))
			default:
				l.Error(ctx, info.FullMethod, logger.Duration(DurationKey, duration), logger.Error(err))
				// GatewayWrongFieldError сюда же
			}
		} else {
			l.Info(ctx, info.FullMethod, logger.Duration(DurationKey, duration))
		}

		err = toGRPCError(err) // значение меняется на статус код, для return

		return res, err
	}

func toGRPCError(err error) error {
	if err == nil {
		return nil
	}

	var ve *errs.ValidationError
	if errors.As(err, &ve) {
		return status.Error(codes.InvalidArgument, ve.Reason)
	}
	if errors.Is(err, context.DeadlineExceeded) {
		return status.Error(codes.DeadlineExceeded, err.Error())
	}
	if errors.Is(err, context.Canceled) {
		return status.Error(codes.Canceled, err.Error())
	}
	if errors.Is(err, errs.ErrNotFound) {
		return status.Error(codes.NotFound, errs.ErrNotFound.Error())
	}
	if errors.Is(err, errs.ErrAlreadyExists) {
		return status.Error(codes.AlreadyExists, errs.ErrAlreadyExists.Error())
	}
	if errors.Is(err, errs.ErrUserPatchEmpty) {
		return status.Error(codes.InvalidArgument, errs.ErrUserPatchEmpty.Error())
	}
	if errors.Is(err, errs.ErrInvalidCredentials) {
		return status.Error(codes.Unauthenticated, errs.ErrInvalidCredentials.Error())
	}
	if errors.Is(err, errs.ErrNameAlreadyExists) { // for auth
		return status.Error(codes.AlreadyExists, errs.ErrNameAlreadyExists.Error())
	}
	if errors.Is(err, errs.ErrEmailAlreadyExists) { // for auth
		return status.Error(codes.AlreadyExists, errs.ErrEmailAlreadyExists.Error())
	}
	if errors.Is(err, errs.ErrPermissionDenied) {
		return status.Error(codes.PermissionDenied, errs.ErrPermissionDenied.Error())
	}
	if errors.Is(err, errs.ErrPasswordMismatch) {
		return status.Error(codes.Unauthenticated, errs.ErrPasswordMismatch.Error())
	}
	if errors.Is(err, errs.ErrWrongMethod) {
		return status.Error(codes.InvalidArgument, errs.ErrWrongMethod.Error())
	}
	if errors.Is(err, errs.ErrBannedUser) {
		return status.Error(codes.PermissionDenied, errs.ErrBannedUser.Error())
	}
	return status.Error(codes.Internal, errs.ErrInternal.Error())  
	// GatewayWrongFieldError сюда же
}

	}

	var logger *zap.Logger

	switch cfg.Level {
	case "dev", "development", "local":
		logger, err = zap.NewDevelopment()
	case "prod", "production":
		logger, err = zap.NewProduction()
	default:
		logger, err = zap.NewProduction()

	}

		return &L{z: logger}, nil
	}

type TokenManager interface {
	Parse(tokenString string) (*Claims, error) // for API-gateway
}

type Manager struct {
	signingKey      string
	accessTokenTTL  time.Duration
	refreshTokenTTL time.Duration
}

type Claims struct {
	UserID  uuid.UUID          `json:"user_id"`
	Role    domain.Role        `json:"role"`
	Premium domain.PremiumType `json:"premium"`
	jwt.RegisteredClaims
}

func NewManager(signingKey string, accessTokenTTL time.Duration, refreshTokenTTL time.Duration) *Manager {
	return &Manager{
		signingKey:      signingKey,
		accessTokenTTL:  accessTokenTTL,
		refreshTokenTTL: refreshTokenTTL,
	}
}

func (m *Manager) Parse(tokenString string) (*Claims, error) {
	var claims Claims
	token, err := jwt.ParseWithClaims(tokenString, &claims, func(token *jwt.Token) (interface{}, error) {
		if token.Method != jwt.SigningMethodHS256 {
			return nil, fmt.Errorf("unexpected signing method: %v", token.Header["alg"])
		}
		return []byte(m.signingKey), nil
	},
		jwt.WithValidMethods([]string{"HS256"}),
		jwt.WithExpirationRequired(),
	)
	if err != nil {
		return nil, err
	}
	if !token.Valid {
		return nil, fmt.Errorf("invalid token")
	}
	return &claims, nil
}

func New(ctx context.Context) (*App, error) {
	cfg, err := torrentconfig.Load()

		return nil, fmt.Errorf("init config: %w", err)
	}

	l, err := logger.New(&cfg.Logger)
	if err != nil {

	}

	// создаёт пул и делает Ping
	pool, err := db.NewPool(&cfg.Postgres)

	svc := service.NewTorrentService(repo)
	handler := transport.NewServer(svc)

	grpcServer := grpc.NewServer(

			interceptors.RecoveryInterceptor(l),
			interceptors.LoggingInterceptor(l),
			interceptors.TimeoutInterceptor(&cfg.Grpc),
		))
	pb.RegisterTorrentServiceServer(grpcServer, handler)

	httpServer := infraServerInit(pool, &cfg.Infra, l)

	return &App{
		cfg:        cfg,
		l:          l,
		pool:       pool,
		grpcServer: grpcServer,
		httpServer: httpServer,
	}, nil

func (a *App) Run(ctx context.Context) error {
	logCtx := context.WithoutCancel(ctx)

	g, gctx := errgroup.WithContext(ctx)

	// gRPC-сервер
	g.Go(func() error {
		lis, err := net.Listen("tcp", fmt.Sprintf(":%d", a.cfg.Grpc.Port))
		if err != nil {
			return fmt.Errorf("grpc listen: %w", err)
		}
		a.l.Info(logCtx, "grpc server starting", logger.Int("port", a.cfg.Grpc.Port))

		err = a.grpcServer.Serve(lis)
		if errors.Is(err, grpc.ErrServerStopped) {
			return nil
		}
		return err
	})

	// http-сервер (health, ready, metrics)
	g.Go(func() error {
		a.l.Info(logCtx, "http infra server starting")

		err := a.httpServer.ListenAndServe()
		if errors.Is(err, http.ErrServerClosed) {
			return nil // после Shutdown это не ошибка
		}
		return err
	})

	// Сторож: спит, пока не придёт сигнал или не упадёт один из серверов
	g.Go(func() error {
		<-gctx.Done()
		return a.shutdown(logCtx)
	})

	if err := g.Wait(); err != nil {
		a.l.Error(logCtx, "torrent-service stopped with error", logger.Error(err))
		return err
	}
	a.l.Info(logCtx, "torrent-service stopped gracefully")
	return nil
}

// shutdown останавливает оба сервера одновременно
func (a *App) shutdown(ctx context.Context) error {
	a.l.Info(ctx, "shutting down")

	timeout, cancel := context.WithTimeout(context.Background(), a.cfg.GracefulShutdownTimeout)
	defer cancel()

	grpcStopped := make(chan struct{})
	httpStopped := make(chan struct{})
	allStopped := make(chan struct{})
	

	go func() {
		defer close(grpcStopped)
		a.grpcServer.GracefulStop()

	}()
	var httpErr error
	go func() {
		defer close(httpStopped)
		httpErr = a.httpServer.Shutdown(timeout)
	}()
	go func() {
		<-grpcStopped
		<-httpStopped
		close(allStopped)
	}()

	select {
	case <-allStopped:
		if httpErr != nil {
			a.httpServer.Close()
			return fmt.Errorf("http shutdown: %w", httpErr)
		}
		a.l.Info(ctx, "grpc and http servers stopped gracefully")
		return nil
	case <-timeout.Done():
		a.l.Error(ctx, "shutdown timed out, forcing stop")
		a.grpcServer.Stop()
		a.httpServer.Close()
		return errors.New("shutdown timeout: servers were stopped forcibly")
	}
}

func (a *App) Close() error {
	a.pool.Close()

	a.l.Sync()

	return nil

func infraServerInit(pool *pgxpool.Pool, cfg *config.Infra, l logger.Logger) *http.Server {
	mux := http.NewServeMux()

	mux.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(http.StatusOK)
		w.Write([]byte("OK"))
	})

	mux.HandleFunc("/ready", func(w http.ResponseWriter, r *http.Request) {

		ctx, cancel := context.WithTimeout(r.Context(), cfg.Timeout/2)
		defer cancel()

		if err := pool.Ping(ctx); err != nil {
			w.WriteHeader(http.StatusServiceUnavailable)
			w.Write([]byte("ERROR"))
			l.Error(ctx, "error on connection to db", logger.Error(err))
			
		} else {
			w.WriteHeader(http.StatusOK)
			w.Write([]byte("OK"))
		}
	})

	mux.Handle("/metrics", promhttp.Handler())

	return &http.Server{
		Addr:    fmt.Sprintf(":%d", cfg.Port),
		Handler: mux,
		ReadHeaderTimeout: cfg.Timeout,
		WriteTimeout:      cfg.Timeout,
		ReadTimeout:       cfg.Timeout,
		IdleTimeout:       cfg.Timeout*6,
	}
}

	}
	l := slog.New(slog.NewTextHandler(os.Stdout, nil))

	dsn := db.BuildDSN(&cfg.Postgres)

	m, err := migrate.New("file://migrations", dsn)
	if err != nil {
		l.Error("error creating migration", slog.String("error", err.Error()))
		os.Exit(1)
	}

	if len(os.Args) < 2 {
		l.Error("add command")
		os.Exit(1)
	}
	command := os.Args[1]
	switch command {
	case "up":
		err = m.Up()
	case "down":
		err = m.Down()
	case "step":
		if len(os.Args) > 2 && os.Args[2] != "" {
			steps, parseErr := strconv.Atoi(os.Args[2])
			if parseErr != nil {
				l.Error("error converting steps to int", slog.String("error", parseErr.Error()))
				os.Exit(1)
			}
			err = m.Steps(steps)

			l.Error("add number of steps")
			os.Exit(1)
		}

		l.Error("wrong command")
		os.Exit(1)
	}

		if errors.Is(err, migrate.ErrNoChange) {
			l.Info("database is already up")

			os.Exit(1)
		}
	}

type Manager struct {
	signingKey      string
	accessTokenTTL  time.Duration
	refreshTokenTTL time.Duration
}

type Claims = jwtparser.Claims

func NewManager(signingKey string, accessTokenTTL time.Duration, refreshTokenTTL time.Duration) *Manager {
	return &Manager{
		signingKey:      signingKey,
		accessTokenTTL:  accessTokenTTL,
		refreshTokenTTL: refreshTokenTTL,
	}
}

func (m *Manager) NewToken(userID uuid.UUID, role domain.Role, premium domain.PremiumType) (string, error) {
	t := jwt.NewWithClaims(jwt.SigningMethodHS256, Claims{
		UserID:  userID,
		Role:    role,
		Premium: premium,
		RegisteredClaims: jwt.RegisteredClaims{
			ExpiresAt: jwt.NewNumericDate(time.Now().Add(m.accessTokenTTL)),
			IssuedAt:  jwt.NewNumericDate(time.Now()),
		},
	})

	token, err := t.SignedString([]byte(m.signingKey))
	if err != nil {
		return "", err
	}
	return token, nil
}

func (m *Manager) NewRefreshToken() (string, error) {
	b := make([]byte, 32)

	_, err := rand.Read(b)
	if err != nil {
		return "", err
	}
	return fmt.Sprintf("%x", b), nil
}

func (m *Manager) RefreshTTL() time.Duration {
	return m.refreshTokenTTL

func New(ctx context.Context) (*App, error) {
	cfg, err := authconfig.Load()
	if err != nil {
		return nil, fmt.Errorf("init config: %w", err)
	}

	l, err := logger.New(&cfg.Logger)
	if err != nil {
		return nil, fmt.Errorf("init logger: %w", err)
	}
	l.Info(ctx, "config & logger init complete")

	// создаёт пул и делает Ping
	pool, err := db.NewPool(&cfg.Postgres)
	if err != nil {
		return nil, fmt.Errorf("create db pool: %w", err)
	}
	l.Info(ctx, "db pool init complete")

	rclient, err := redisclient.NewClient(ctx, &cfg.Redis)
	if err != nil {
		return nil, fmt.Errorf("create redis client: %w", err)
	}
	l.Info(ctx, "redis client init complete")

	urepo := repository.NewPostgresRepository(pool)
	rfrepo := repository.NewRedisRepository(rclient)
	m := jwtmanager.NewManager(cfg.JWT.SigningKey, cfg.JWT.AccessTokenTTL, cfg.JWT.RefreshTokenTTL)
	h := hasher.NewSimpleHasher(cfg.Hasher.Cost)
	
	svc := service.New(m, h, urepo, rfrepo)
	handler := transport.NewServer(svc)

	grpcServer := grpc.NewServer(
		grpc.ChainUnaryInterceptor(
			interceptors.RecoveryInterceptor(l),
			interceptors.LoggingInterceptor(l),
			interceptors.TimeoutInterceptor(&cfg.Grpc),
		))
	pb.RegisterAuthServiceServer(grpcServer, handler)

	httpServer := infraServerInit(pool, rclient, &cfg.Infra, l)

	return &App{
		cfg:        cfg,
		l:          l,
		pool:       pool,
		rclient:    rclient,
		grpcServer: grpcServer,
		httpServer: httpServer,
	}, nil
}

func (a *App) Run(ctx context.Context) error {
	logCtx := context.WithoutCancel(ctx) // контекст для логгера (останется после отмены основного через gracefulShutdown)

	g, gctx := errgroup.WithContext(ctx)

	// gRPC-сервер

		lis, err := net.Listen("tcp", fmt.Sprintf(":%d", a.cfg.Grpc.Port))
		if err != nil {
			return fmt.Errorf("grpc listen: %w", err)
		}
		a.l.Info(logCtx, "grpc server starting", logger.Int("port", a.cfg.Grpc.Port))

		err = a.grpcServer.Serve(lis)
		if errors.Is(err, grpc.ErrServerStopped) {
			return nil
		}
		return err
	})

	// http-сервер (health, ready, metrics)
	g.Go(func() error {
		a.l.Info(logCtx, "http infra server starting")

		err := a.httpServer.ListenAndServe()
		if errors.Is(err, http.ErrServerClosed) {
			return nil // после Shutdown это не ошибка
		}
		return err
	})

	// Сторож: спит, пока не придёт сигнал или не упадёт один из серверов
	g.Go(func() error {
		<-gctx.Done()
		return a.shutdown(logCtx)
	})

	if err := g.Wait(); err != nil {
		a.l.Error(logCtx, "auth-service stopped with error", logger.Error(err))
		return err
	}
	a.l.Info(logCtx, "auth-service stopped gracefully")
	return nil
}

// shutdown останавливает оба сервера одновременно
func (a *App) shutdown(ctx context.Context) error {
	a.l.Info(ctx, "shutting down")

	timeout, cancel := context.WithTimeout(context.Background(), a.cfg.GracefulShutdownTimeout)
	defer cancel()

	grpcStopped := make(chan struct{})
	httpStopped := make(chan struct{})
	allStopped := make(chan struct{})
	

	go func() {
		defer close(grpcStopped)
		a.grpcServer.GracefulStop()

	}()
	var httpErr error
	go func() {
		defer close(httpStopped)
		httpErr = a.httpServer.Shutdown(timeout)
	}()
	go func() {
		<-grpcStopped
		<-httpStopped
		close(allStopped)
	}()

	select {
	case <-allStopped:
		if httpErr != nil {
			a.httpServer.Close()
			return fmt.Errorf("http shutdown: %w", httpErr)
		}
		a.l.Info(ctx, "grpc and http servers stopped gracefully")
		return nil
	case <-timeout.Done():
		a.l.Error(ctx, "shutdown timed out, forcing stop")
		a.grpcServer.Stop()
		a.httpServer.Close()
		return errors.New("shutdown timeout: servers were stopped forcibly")
	}
}

func (a *App) Close() error {
	a.pool.Close()
	a.rclient.Close()

	a.l.Sync()

	return nil

func (s *UserService) Login(ctx context.Context, email, password string) (access, refresh string, err error) {
	const op = "service.User.Login"
	if err := validateEmail(email); err != nil {
		return "", "", fmt.Errorf("%s: %w", op, err)
	}
	u, err := s.urepo.GetUserByEmail(ctx, email)
	if err != nil {
		if errors.Is(err, errs.ErrNotFound) {
			return "", "", fmt.Errorf("%s: %w", op, errs.ErrInvalidCredentials) // не раскрываем
		}
		return "", "", fmt.Errorf("%s: %w", op, err)
	}
	if err = s.h.CheckPassword(password, u.PasswordHash); err != nil {
		if errors.Is(err, errs.ErrPasswordMismatch) {
			return "", "", fmt.Errorf("%s: %w", op, errs.ErrInvalidCredentials)
		}
		return "", "", fmt.Errorf("%s: %w", op, err)
	}
	if u.Role == models.BannedRole {
		return "", "", fmt.Errorf("%s: %w", op, errs.ErrBannedUser)
	}


	

	access, err = s.m.NewToken(u.ID, u.Role, u.Premium)
	if err != nil {
		return "", "", fmt.Errorf("%s: %w", op, err)
	}

	refresh, err = s.m.NewRefreshToken()
	if err != nil {
		return "", "", fmt.Errorf("%s: %w", op, err)
	}

	err = s.rfrepo.SetRefreshToken(ctx, hashToken(refresh), u.ID.String(), s.m.RefreshTTL())
	if err != nil {
		return "", "", fmt.Errorf("%s: %w", op, err)
	}

	return access, refresh, nil
}

func (s *UserService) Refresh(ctx context.Context, rawRefresh string) (access, newRefresh string, err error) {
    const op = "service.User.Refresh"

    if rawRefresh == "" {
        return "", "", fmt.Errorf("%s: %w", op, &errs.ValidationError{Field: "refresh_token", Reason: "required"})
    }

    newRaw, err := s.m.NewRefreshToken()
    if err != nil {
        return "", "", fmt.Errorf("%s: %w", op, err)
    }

    userIDStr, err := s.rfrepo.RotateRefreshToken(
        ctx, hashToken(rawRefresh), hashToken(newRaw), s.m.RefreshTTL(),
    )
    if err != nil {
		if errors.Is(err, errs.ErrNotFound) {
			return "", "", fmt.Errorf("%s: %w", op, errs.ErrInvalidCredentials) // → 401
		}
        return "", "", fmt.Errorf("%s: %w", op, err)
    }

    userID, err := uuid.Parse(userIDStr)
    if err != nil {
		_ = s.rfrepo.DeleteRefreshToken(ctx, hashToken(newRaw))
        return "", "", fmt.Errorf("%s: %w", op, err)
    }

    u, err := s.urepo.GetUserPrivate(ctx, userID) // актуальные role/premium
    if err != nil {
		_ = s.rfrepo.DeleteRefreshToken(ctx, hashToken(newRaw))
        return "", "", fmt.Errorf("%s: %w", op, err)
    }

	if u.Role == models.BannedRole {
		_ = s.rfrepo.DeleteRefreshToken(ctx, hashToken(newRaw))
		return "", "", fmt.Errorf("%s: %w", op, errs.ErrBannedUser)
	}

    access, err = s.m.NewToken(u.ID, u.Role, u.Premium)
    if err != nil {
		_ = s.rfrepo.DeleteRefreshToken(ctx, hashToken(newRaw))
        return "", "", fmt.Errorf("%s: %w", op, err)
    }

    return access, newRaw, nil
}
func hashToken(token string) string {
	h := sha256.Sum256([]byte(token))
	return hex.EncodeToString(h[:])

	case models.BannedRole:
		return false, fmt.Errorf("%s: %w", op, errs.ErrWrongMethod)
	}

		return false, fmt.Errorf("%s: %w", op, errs.ErrPermissionDenied)
	}

	updated, err = s.urepo.UpdateUserRole(ctx, id, TargetRole)
	if err != nil {
		return false, fmt.Errorf("%s: %w", op, err)
	}

package repository

import (
	"context"
	"fmt"
	"math"
	"time"

	"github.com/monstrong/gracker2/pkg/errs"
	"github.com/redis/go-redis/v9"

const userRefreshTokensPrefix = "user-refresh-tokens:"
const refreshPrefix = "refresh:"

type RedisRepository struct {
	c redis.UniversalClient
}
func NewRedisRepository(c redis.UniversalClient) *RedisRepository {
	return &RedisRepository{
		c: c,
	}
}
// redis выглядит так:
// refresh:hash_A             → "123"
// refresh:hash_B             → "123"

// user-refresh-tokens:123    → Set{"hash_A", "hash_B"}

// стоит помнить что в SET могут лежать ключи ныне не существующие. нет их автоматического удаления когда сам ключ refresh:hash_A сгорает
// Поскольку при ротации продлевается TTL множества, оно может продолжать существовать очень долго, а устаревшие хеши будут занимать память
// Однако при revokeAllRefreshTokens они удалятся. по истечении ttl самого множества,
// то есть когда пользователь вообще не будет заходить в аккаунт на протяжении месяца ни с одного устройства.
// возможное РЕШЕНИЕ - периодический проход по всей бд и удаление устаревших хэшей из set'ов. 

// При Redis Cluster важно глобально пересмотреть логику работы, так как скрипт берет сразу два ключа, а они могут храниться в разных частях

func refreshKey(tokenHash string) string {
	return refreshPrefix + tokenHash
}

func userRefreshTokensKey(userID string) string {
	return userRefreshTokensPrefix + userID
}

var deleteRefreshScript = redis.NewScript(`
	local tokenKey = KEYS[1]
	local userRefreshTokensPrefix = ARGV[1]
	local tokenHash = ARGV[2]

	local userID = redis.call("GET", tokenKey)

	if userID then
		local sessionsKey = userRefreshTokensPrefix .. userID
		redis.call("SREM", sessionsKey, tokenHash)
	end

	redis.call("DEL", tokenKey)

	return 1
`)

var rotateRefreshScript = redis.NewScript(`
	local oldTokenKey = KEYS[1]
	local newTokenKey = KEYS[2]

	local ttlSeconds = tonumber(ARGV[1])
	local userRefreshTokensPrefix = ARGV[2]
	local oldTokenHash = ARGV[3]
	local newTokenHash = ARGV[4]

	local userID = redis.call("GET", oldTokenKey)

	if not userID then
		return ""
	end

	redis.call("SET", newTokenKey, userID, "EX", ttlSeconds)

	local sessionsKey = userRefreshTokensPrefix .. userID

	redis.call("SREM", sessionsKey, oldTokenHash)
	redis.call("SADD", sessionsKey, newTokenHash)
	redis.call("EXPIRE", sessionsKey, ttlSeconds)

	redis.call("DEL", oldTokenKey)

	return userID
`)

var revokeAllRefreshTokensScript = redis.NewScript(`
	local sessionsKey = KEYS[1]
	local refreshPrefix = ARGV[1]

	local hashes = redis.call("SMEMBERS", sessionsKey)

	for _, hash in ipairs(hashes) do
		redis.call("DEL", refreshPrefix .. hash)
	end

	redis.call("DEL", sessionsKey)

	return #hashes
`)

func (r *RedisRepository) SetRefreshToken(
	ctx context.Context,
	tokenHash string,
	userID string,
	ttl time.Duration,
) error {
	const op = "repo.Redis.SetRefreshToken"

	if ttl <= 0 {
		return fmt.Errorf("%s: ttl must be positive", op)
	}

	_, err := r.c.TxPipelined(ctx, func(pipe redis.Pipeliner) error {
		pipe.Set(ctx, refreshKey(tokenHash), userID, ttl)

		sessionsKey := userRefreshTokensKey(userID)
		pipe.SAdd(ctx, sessionsKey, tokenHash)
		pipe.Expire(ctx, sessionsKey, ttl)

		return nil
	})
	if err != nil {
		return fmt.Errorf("%s: %w", op, err)
	}

	return nil
}

func (r *RedisRepository) DeleteRefreshToken(
	ctx context.Context,
	tokenHash string,
) error {
	const op = "repo.Redis.DeleteRefreshToken"

	err := deleteRefreshScript.Run(
		ctx,
		r.c,
		[]string{refreshKey(tokenHash)},
		userRefreshTokensPrefix,
		tokenHash,
	).Err()
	if err != nil {
		return fmt.Errorf("%s: %w", op, err)
	}

	return nil
}

func (r *RedisRepository) RotateRefreshToken(
	ctx context.Context,
	oldTokenHash string,
	newTokenHash string,
	ttl time.Duration,
) (string, error) {
	const op = "repo.Redis.RotateRefreshToken"

	if ttl <= 0 {
		return "", fmt.Errorf("%s: ttl must be positive", op)
	}

	ttlSeconds := int64(math.Ceil(ttl.Seconds()))

	userID, err := rotateRefreshScript.Run(
		ctx,
		r.c,
		[]string{
			refreshKey(oldTokenHash),
			refreshKey(newTokenHash),
		},
		ttlSeconds,
		userRefreshTokensPrefix,
		oldTokenHash,
		newTokenHash,
	).Text()
	if err != nil {
		return "", fmt.Errorf("%s: %w", op, err)
	}

	if userID == "" {
		return "", fmt.Errorf("%s: %w", op, errs.ErrNotFound)
	}

	return userID, nil
}

func (r *RedisRepository) RevokeAllUserRefreshTokens(
	ctx context.Context,
	userID string,
) error {
	const op = "repo.Redis.RevokeAllUserRefreshTokens"

	err := revokeAllRefreshTokensScript.Run(
		ctx,
		r.c,
		[]string{userRefreshTokensKey(userID)},
		refreshPrefix,
	).Err()
	if err != nil {
		return fmt.Errorf("%s: %w", op, err)
	}

	return nil
}

func checkIDfromMD(MD metadata.MD) (userID uuid.UUID, err error) {
	vals := MD.Get(grpcmetakey.GRPCMetaKeyUserID)

	if len(vals) > 0 {
		if userID, err = uuid.Parse(vals[0]); err != nil {
			return uuid.Nil, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyUserID, Reason: "invalid user id format"}
		}
	} else {
		return uuid.Nil, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyUserID, Reason: "empty user id"}
	}

	return
}

func checkSenderRolefromMD(MD metadata.MD) (SenderRole models.Role, err error) {
	vals := MD.Get(grpcmetakey.GRPCMetaKeyRole)

	if len(vals) > 0 {
		roleValue, err := strconv.ParseInt(vals[0], 10, 8)
		if err != nil {
			return models.RoleUnspecified, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyRole, Reason: "invalid role format"}
		}
		SenderRole = models.Role(roleValue)
	} else {
		return models.RoleUnspecified, &errs.GatewayWrongFieldError{Field: grpcmetakey.GRPCMetaKeyRole, Reason: "empty role"}
	}

	return
}

package authconfig

import (
	"fmt"
	"os"
	"time"

	"github.com/monstrong/gracker2/pkg/config"
	"golang.org/x/crypto/bcrypt"
)

type Config struct {
	config.Config `mapstructure:",squash"`

	JWT    JWT    `mapstructure:"jwt" yaml:"jwt"`
	Hasher Hasher `mapstructure:"hasher" yaml:"hasher"`
	Redis  Redis  `mapstructure:"redis" yaml:"redis"`
}

type JWT struct {
	AccessTokenTTL  time.Duration `mapstructure:"access_token_ttl" yaml:"access_token_ttl"`
	RefreshTokenTTL time.Duration `mapstructure:"refresh_token_ttl" yaml:"refresh_token_ttl"`
	SigningKey      string
}

type Redis struct {
    Addr         string        `mapstructure:"addr"`
    Password     string        
    DB           int           `mapstructure:"db"`
    DialTimeout  time.Duration `mapstructure:"dial_timeout"`
    ReadTimeout  time.Duration `mapstructure:"read_timeout"`
    WriteTimeout time.Duration `mapstructure:"write_timeout"`
}

type Hasher struct {
	Cost int `mapstructure:"cost" yaml:"cost"`
}

const secretKey = "JWT_SECRET"
const postgresPasswordKey = "POSTGRES_PASSWORD"
const redisPasswordKey = "REDIS_PASSWORD"

func Load() (*Config, error) {
	cfg := &Config{}

	if err := config.LoadInto(cfg); err != nil {
		return nil, err
	}

	cfg.Postgres.Password = os.Getenv(postgresPasswordKey)
	cfg.JWT.SigningKey = os.Getenv(secretKey)
	cfg.Redis.Password = os.Getenv(redisPasswordKey)

	if err := config.Validate(&cfg.Config); err != nil {
		return nil, err
	}

	if cfg.JWT.SigningKey == "" {
		return nil, fmt.Errorf("empty JWT secret")
	}
	if len(cfg.JWT.SigningKey) < 32 {
		return nil, fmt.Errorf("invalid JWT secret length")
	}
	if cfg.JWT.AccessTokenTTL <= 0 {
		return nil, fmt.Errorf("invalid access token TTL")
	}
	if cfg.JWT.RefreshTokenTTL <= 0 {
		return nil, fmt.Errorf("invalid refresh token TTL")
	}
	
	if cfg.Hasher.Cost < bcrypt.MinCost ||
		cfg.Hasher.Cost > bcrypt.MaxCost {
		return nil, fmt.Errorf("invalid bcrypt cost: %d", cfg.Hasher.Cost)
	}

	return cfg, nil
}
